cmake_minimum_required(VERSION 3.16)
project(camera_inference LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(camera_inference SHARED detector.cpp)
target_compile_features(camera_inference PRIVATE cxx_std_17)
target_link_libraries(camera_inference PRIVATE openvino::runtime ${OpenCV_LIBS})
target_include_directories(camera_inference PRIVATE ${OpenCV_INCLUDE_DIRS})
set_target_properties(camera_inference PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS camera_inference LIBRARY DESTINATION lib)
